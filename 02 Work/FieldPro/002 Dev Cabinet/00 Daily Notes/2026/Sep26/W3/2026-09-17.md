## 📝 Yesterday
```dataviewjs
const currentDate = dv.current().file.name;

const yesterday = window.moment(currentDate, "YYYY-MM-DD", true)
    .subtract(1, "day")
    .format("YYYY-MM-DD");

const page = dv.pages("")
    .find(p => p.file.name === yesterday);

if (page) {
    const items = page.file.lists
        .filter(item =>
            item.section?.subpath &&
            item.section.subpath.trim().endsWith("Today")
        );

    if (items.length > 0) {
        dv.list(items.map(item => item.text));
    } else {
        dv.paragraph(`Found ${yesterday}, but no items were found under Today.`);
    }
} else {
    dv.paragraph(`No daily note found for ${yesterday}.`);
}
```
