``` sql
SELECT 
	[login].[TenantID],
	[login].[UserID],
	[login].Name,
	[login].FirstLogin,
	[login].LastLogin,
	[login].Active,
	[login].ActiveInMonth AS 'DeActiveInMonth',
	[user].Email,
	tenant.CompanyName,
	tenant.WorkflowName,
	[user].RoleSuperAdministration,
	[user].RoleSiteAdministration
FROM [vLoginWithAudit] [login]
	INNER JOIN vUser [user] ON [user].ID = login.UserID
	INNER JOIN vTenant tenant ON tenant.ID = login.TenantID
WHERE [login].TenantID = 1 AND [login].[UserID] = 4
ORDER BY login.TenantID, login.UserID
```
